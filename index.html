구글 스프레드시트 연동 설정

운동 기록을 구글 시트와 양방향으로 자동 동기화합니다.

📖 최신 Apps Script 연동 코드

스프레드시트 상단 확장 프로그램 > Apps Script에 코드를 붙여넣고, 배포 > 배포 관리 > [신규 버전]으로 배포하시면 모바일에서도 완벽하게 양방향 동기화됩니다. 
// 1. 구글 시트에서 운동 기록 가져오기 (GET)
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("운동 기록");
    if (!sheet) return ContentService.createTextOutput(JSON.stringify({ result: "error" })).setMimeType(ContentService.MimeType.JSON);
    var displayData = sheet.getDataRange().getDisplayValues();
    var rows = [];
    for (var i = 0; i < displayData.length; i++) {
      var row = displayData[i];
      var d = String(row[0]||"").trim(), ex = String(row[2]||"").trim();
      if (!d || !ex || ex === "운동 종목" || d === "날짜") continue;
      if (/\d{4}/.test(d)) {
        rows.push({
          date: d, time: String(row[1]||"").trim(), exercise: ex,
          setNum: parseInt(row[3],10)||1,
          targetWeight: parseFloat(String(row[4]).replace(/[^0-9.]/g,''))||0,
          targetReps: parseInt(String(row[5]).replace(/[^0-9]/g,''),10)||0,
          actualWeight: parseFloat(String(row[6]).replace(/[^0-9.]/g,''))||0,
          actualReps: parseInt(String(row[7]).replace(/[^0-9]/g,''),10)||0,
          volume: parseFloat(String(row[8]).replace(/[^0-9.]/g,''))||0,
          duration: String(row[9]||"").trim(), notes: String(row[10]||"").trim()
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ result: "success", rows: rows })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
// 2. 구글 시트에 새 운동 기록 추가 (POST - 중복 방지 강화)
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet(), sheet = ss.getSheetByName("운동 기록");
    var data = JSON.parse(e.postData.contents);
    if (data.rows && data.rows.length > 0) {
      var displayData = sheet.getDataRange().getDisplayValues();
      var existingKeys = {}, lastRow = 7;
      
      // 날짜/운동명/세트번호 정규화 키 생성 (셀 서식과 무관하게 100% 중복 차단)
      function makeNormKey(dVal, exVal, setVal) {
        var cleanDate = String(dVal || '').replace(/[^0-9]/g, '').substring(0, 8);
        var cleanEx = String(exVal || '').replace(/\s+/g, '');
        var cleanSet = parseInt(setVal, 10) || 1;
        return cleanDate + '__' + cleanEx + '__' + cleanSet;
      }

      for (var i = 0; i < displayData.length; i++) {
        var r = displayData[i];
        if (r[0] && /\d{4}/.test(r[0]) && r[2] && r[2] !== "운동 종목") {
          lastRow = i + 1;
          existingKeys[makeNormKey(r[0], r[2], r[3])] = true;
        }
      }

      data.rows.forEach(function(row) {
        var k = makeNormKey(row[0], row[2], row[3]);
        if (!existingKeys[k]) {
          lastRow++;
          sheet.getRange(lastRow, 1, 1, row.length).setValues([row]);
          sheet.getRange(lastRow, 9).setFormula("=G" + lastRow + "*H" + lastRow);
          existingKeys[k] = true;
        }
      });
    }
    return ContentService.createTextOutput(JSON.stringify({ result: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
